Ниже пример того, как можно “атомарно” собрать весь экран в React-компонентах со styled-components, включая все состояния у кнопок и карточки с “ошибкой”:

```jsx
// App.jsx
import React, { useState } from 'react'
import styled, { css } from 'styled-components'
import { Arrow } from './icons'

// ——— Атомарные компоненты ———

const Button = styled.button`
  display: inline-flex;
  align-items: center;
  justify-content: center;
  padding: 12px 20px;
  font-size: 16px;
  border-radius: 8px;
  border: none;
  cursor: pointer;
  transition: all 0.2s;
  ${(p) =>
    p.primary
      ? css`
          background: linear-gradient(90deg, #4facfe 0%, #00f2fe 100%);
          color: #fff;
          &:hover:not(:disabled) {
            background: linear-gradient(90deg, #3facec 0%, #00d2ec 100%);
          }
        `
      : css`
          background: #f0f4f8;
          color: #333;
          &:hover:not(:disabled) {
            background: #e2e8f0;
          }
        `}

  &:disabled {
    background: #ddd;
    color: #999;
    cursor: default;
  }
  &:focus {
    outline: 2px solid #4facfe;
    outline-offset: 2px;
  }
`

const InputWrapper = styled.div`
  display: flex;
  flex-direction: column;
`
const Label = styled.label`
  font-size: 14px;
  color: #555;
  margin-bottom: 4px;
`
const InputField = styled.input`
  padding: 10px;
  font-size: 16px;
  border: 1px solid #ccc;
  border-radius: 6px;
  &:focus {
    border-color: #4facfe;
    outline: none;
  }
`
const Suffix = styled.span`
  position: absolute;
  right: 12px;
  top: 50%;
  transform: translateY(-50%);
  color: #888;
`

export function Input({ label, suffix, ...props }) {
  return (
    <InputWrapper>
      {label && <Label>{label}</Label>}
      <div style={{ position: 'relative' }}>
        <InputField {...props} />
        {suffix && <Suffix>{suffix}</Suffix>}
      </div>
    </InputWrapper>
  )
}

const Section = styled.section`
  background: #fff;
  border-radius: 16px;
  padding: 24px;
  margin-bottom: 24px;
`
const TitleRow = styled.div`
  display: flex;
  align-items: center;
  justify-content: space-between;
  margin-bottom: 16px;
`
const Title = styled.h2`
  margin: 0;
  font-size: 20px;
  color: #333;
`

// ——— Специальные компоненты экрана ———

const FolderButton = styled(Button)`
  width: 100%;
  justify-content: space-between;
`

const Fields = styled.div`
  display: flex;
  gap: 16px;
  margin-bottom: 24px;
`
const CardGrid = styled.div`
  display: grid;
  grid-template-columns: repeat(auto-fill, minmax(200px, 1fr));
  gap: 16px;
`

const CardContainer = styled.div`
  display: flex;
  background: #f9fafb;
  border-radius: 12px;
  padding: 16px;
  border: 1px solid #ddd;
  ${(p) =>
    p.error &&
    css`
      border-color: #f44336;
    `}
`
const Preview = styled.div`
  width: 80px;
  height: 80px;
  background: #dfffe0;
  border-radius: 8px;
  flex-shrink: 0;
  margin-right: 16px;
`
const Info = styled.div`
  display: flex;
  flex-direction: column;
`
const Name = styled.div`
  font-size: 16px;
  font-weight: 500;
  margin-bottom: 8px;
`
const Meta = styled.div`
  font-size: 14px;
  color: #555;
  margin-bottom: 4px;
`

function Card({ name, size, colors, duration, error }) {
  return (
    <CardContainer error={error}>
      <Preview />
      <Info>
        <Name>{name}</Name>
        <Meta>Вес: {size} Кб</Meta>
        <Meta>Цвета: {colors}</Meta>
        <Meta>Длительность: {duration} сек</Meta>
        {error && <Meta style={{ color: '#f44336' }}>⚠️ Ошибка: слишком маленький размер</Meta>}
      </Info>
    </CardContainer>
  )
}

// ——— Основной App ———

const Container = styled.div`
  max-width: 800px;
  margin: 40px auto;
  padding: 0 16px;
`

export default function App() {
  const [folder, setFolder] = useState('')
  const [frameLength, setFrameLength] = useState(3)
  const [maxWeight, setMaxWeight] = useState(500)
  const [cards, setCards] = useState([])

  const handleSelectFolder = () => {
    // тут логика выбора папки
    setFolder('YYsalesCompany_24')
  }
  const handleReset = () => {
    setFrameLength(3)
    setMaxWeight(500)
  }
  const handleConvert = () => {
    // имитация ответа
    setCards([
      { id: 1, name: 'gif1', size: 480, colors: 16, duration: 3, error: false },
      { id: 2, name: 'gif2', size: 300, colors: 8, duration: 3, error: true }, // ошибка
      // …
    ])
  }

  return (
    <Container>
      {/* Данные */}
      <Section>
        <TitleRow>
          <Title>Данные</Title>
        </TitleRow>
        <FolderButton onClick={handleSelectFolder}>
          {folder || 'Выбор папки'}
          <Arrow />
        </FolderButton>
      </Section>

      {/* Настройки */}
      <Section>
        <TitleRow>
          <Title>Настройки</Title>
          <Button onClick={handleReset}>сбросить</Button>
        </TitleRow>
        <Fields>
          <Input
            label="Длина кадра"
            type="number"
            value={frameLength}
            onChange={(e) => setFrameLength(Number(e.target.value))}
            suffix="сек"
          />
          <Input
            label="Максимальный вес"
            type="number"
            value={maxWeight}
            onChange={(e) => setMaxWeight(Number(e.target.value))}
            suffix="Кб"
          />
        </Fields>
        <Button primary disabled={!folder} onClick={handleConvert}>
          Жмахнуть ЖИФ
          <Arrow />
        </Button>
      </Section>

      {/* Результаты */}
      {cards.length > 0 && (
        <Section>
          <TitleRow>
            <Title>Сконвертировано {cards.length} гифов</Title>
            <Button onClick={() => {/* открыть папку */}}>Открыть папку<Arrow /></Button>
          </TitleRow>
          <CardGrid>
            {cards.map((c) => (
              <Card key={c.id} {...c} />
            ))}
          </CardGrid>
        </Section>
      )}
    </Container>
  )
}
```

Иконку стрелки можно сделать так:

```jsx
// icons.jsx
import React from 'react'

export const Arrow = () => (
  <svg width="16" height="16" viewBox="0 0 24 24" fill="none">
    <path d="M8 5l8 7-8 7" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"/>
  </svg>
)
```

**Что тут есть:**

1. **Атомарные компоненты**: `Button`, `Input`, `Card` и т.д.
2. **Стили состояний**:

   * `Button`: hover, focus, active (transform), disabled.
   * `Card` умеет принимать проп `error` и красит рамку / текст в красный.
3. **Структура экрана**:

   * Секция “Данные” с выбором папки.
   * Секция “Настройки” с полями и кнопкой.
   * Секция “Результаты” с гридом карточек.
4. **Раздельные компоненты** позволяют легко поддерживать и расширять интерфейс.

Можно прикрутить любую логику выбора папки, загрузки/конвертации и т.п. – структура уже готова.
